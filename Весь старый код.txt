        // Извлечение вопросов по маркерам
        public static Dictionary<string, List<string>> ExtractQuestions(string filePath, List<string> markers)
        {
            var questions = new Dictionary<string, List<string>>();
            string currentMarker = null;

            using (WordprocessingDocument doc = WordprocessingDocument.Open(filePath, false))
            {
                var body = doc.MainDocumentPart.Document.Body;

                foreach (var paragraph in body.Elements<Paragraph>())
                {
                    string text = paragraph.InnerText.Trim();

                    if (markers.Contains(text))
                    {
                        currentMarker = text;
                        if (!questions.ContainsKey(currentMarker))
                        {
                            questions[currentMarker] = new List<string>();
                        }
                    }
                    else if (currentMarker != null && text.Length > 0)
                    {
                        questions[currentMarker].Add(text);
                    }
                }
            }

            return questions;
        }



        // Генерация билетов
        public static List<List<string>> GenerateTickets(Dictionary<string, List<string>> questions, List<string> usedMarkers, int ticketCount)
        {
            var random = new Random();
            var tickets = new List<List<string>>();

            for (int i = 0; i < ticketCount; i++)
            {
                var ticket = new List<string>();

                foreach (string marker in usedMarkers)
                {
                    if (!questions.ContainsKey(marker) || questions[marker].Count == 0)
                    {
                        ticket.Add("Вопрос отсутствует");
                        continue;
                    }

                    int index = random.Next(questions[marker].Count);
                    string selected = questions[marker][index];
                    ticket.Add($"{marker}: {selected}");

                    // чтобы не повторялся
                    questions[marker].RemoveAt(index);
                }

                tickets.Add(ticket);
            }

            return tickets;
        }





        // Создание документа с билетами
        public static byte[] GenerateTicketsDocument(List<List<string>> tickets, string filledTemplatePath)
        {
            using (MemoryStream outputStream = new MemoryStream())
            {
                using (WordprocessingDocument outputDoc = WordprocessingDocument.Create(outputStream, WordprocessingDocumentType.Document))
                {
                    var mainPart = outputDoc.AddMainDocumentPart();
                    mainPart.Document = new Document(new Body());
                    var outputBody = mainPart.Document.Body;

                    foreach (var (ticket, index) in tickets.Select((t, i) => (t, i)))
                    {
                        using (WordprocessingDocument templateDoc = WordprocessingDocument.Open(filledTemplatePath, false))
                        {
                            var templateBody = (Body)templateDoc.MainDocumentPart.Document.Body.CloneNode(true);

                            var replacements = new Dictionary<string, string>
{
    { "{ticketnumber}", (index + 1).ToString() }
};

                            for (int i = 0; i < ticket.Count; i++)
                            {
                                replacements.Add($"{{QUESTION{i + 1}}}", ticket[i]);
                            }
                            // Удалим все неиспользованные маркеры {QUESTIONX}
                            for (int i = ticket.Count + 1; i <= 4; i++)
                            {
                                string unusedMarker = $"{{QUESTION{i}}}";
                                ReplaceTextPlaceholders(templateBody, new Dictionary<string, string>
    {
        { unusedMarker, "" }
    });
                            }
                            ReplaceTextPlaceholders(templateBody, replacements);

                            foreach (var element in templateBody.Elements())
                                outputBody.Append(element.CloneNode(true));
                        }

                        if (index < tickets.Count - 1)
                        {
                            outputBody.Append(new Paragraph(new Run(new Break() { Type = BreakValues.Page })));
                        }
                    }

                    mainPart.Document.Save();
                }

                return outputStream.ToArray();
            }
        }

        public static bool ValidateQuestionFile(string filePath, List<string> currentRequiredMarkers)
        {
            if (IsFileLocked(filePath))
            {
                MessageBox.Show("Файл уже открыт в другой программе. Пожалуйста, закройте его и попробуйте снова.", "Файл занят", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }
            var actualRecognizedMarkersInDoc = new HashSet<string>();
            try
            {
                using (WordprocessingDocument doc = WordprocessingDocument.Open(filePath, false))
                {
                    var body = doc.MainDocumentPart?.Document?.Body;
                    if (body == null)
                    {
                        MessageBox.Show("Не удалось прочитать структуру документа.", "Ошибка файла", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        return false;
                    }

                    foreach (var paragraph in body.Descendants<Paragraph>())
                    {
                        string text = paragraph.InnerText.Trim();
                        if (AllPossibleQuestionMarkers.Contains(text))
                        {
                            actualRecognizedMarkersInDoc.Add(text);
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Произошла ошибка при чтении файла: {ex.Message}", "Ошибка чтения файла", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }

            
            var requiredMarkersSet = new HashSet<string>(currentRequiredMarkers);

            
            if (actualRecognizedMarkersInDoc.SetEquals(requiredMarkersSet))
            {
                return true; 
            }
            else
            {
                var missingMarkers = requiredMarkersSet.Except(actualRecognizedMarkersInDoc).ToList();
                var extraMarkers = actualRecognizedMarkersInDoc.Except(requiredMarkersSet).ToList();

                StringBuilder errorMessage = new StringBuilder("Файл не соответствует выбранному шаблону билета.");

                if (missingMarkers.Any())
                {
                    errorMessage.Append($"\nОтсутствуют необходимые маркеры: {string.Join(", ", missingMarkers)}");
                }
                if (extraMarkers.Any())
                {
                    errorMessage.Append($"\nОбнаружены лишние маркеры в файле: {string.Join(", ", extraMarkers)}");
                }

                MessageBox.Show(errorMessage.ToString(), "Несоответствие шаблону", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }
        }

        public static int CalculateMaxTickets(Dictionary<string, List<string>> questions, List<string> usedMarkers)
        {
            int min = int.MaxValue;

            foreach (string marker in usedMarkers)
            {
                if (!questions.ContainsKey(marker) || questions[marker].Count == 0)
                    return 0;

                if (questions[marker].Count < min)
                    min = questions[marker].Count;
            }

            return min;
        }

        public static void PrintWordDocumentFromByteArray(byte[] fileContent)
        {
            try
            {
                string tempFilePath = Path.Combine(Path.GetTempPath(), $"{Guid.NewGuid()}.docx");
                File.WriteAllBytes(tempFilePath, fileContent);

                if (!File.Exists(tempFilePath))
                {
                    MessageBox.Show("Файл не найден для печати.");
                    return;
                }

                var psi = new ProcessStartInfo
                {
                    FileName = tempFilePath,
                    Verb = "print",
                    CreateNoWindow = true,
                    WindowStyle = ProcessWindowStyle.Hidden
                };

                using (var process = Process.Start(psi))
                {
                    process.WaitForExit(30000);
                }

                File.Delete(tempFilePath);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка при печати: {ex.Message}");
            }
        }
        public static void UpdateRichTextBox(string template, Dictionary<string, string> replacements, RichTextBox richTextBox)
        {
            string result = template;

            foreach (var pair in replacements)
            {
                string marker = "{" + pair.Key + "}";
                result = result.Replace(marker, pair.Value);
            }

            richTextBox.Text = result;
        }

        public static void ReplaceTextPlaceholders(OpenXmlElement root, Dictionary<string, string> replacements)
        {
            foreach (var paragraph in root.Descendants<Paragraph>())
            {
                var texts = paragraph.Descendants<Text>().ToList();
                string fullText = string.Join("", texts.Select(t => t.Text));

                bool hasPlaceholder = replacements.Keys.Any(marker => fullText.Contains(marker));
                if (!hasPlaceholder) continue;

                foreach (var pair in replacements)
                {
                    fullText = fullText.Replace(pair.Key, pair.Value);
                }

                // Очистим старые Text
                foreach (var text in texts)
                    text.Text = "";

                // Вставим новый текст в первый Run
                var firstRun = paragraph.Elements<Run>().FirstOrDefault();
                if (firstRun != null)
                {
                    firstRun.RemoveAllChildren<Text>();
                    firstRun.AppendChild(new Text(fullText));
                }

                // Центрирование для educationalinstitution
                if (replacements.ContainsKey("{educationalinstitution}") && fullText.Contains(replacements["{educationalinstitution}"]))
                {
                    var props = paragraph.GetFirstChild<ParagraphProperties>();
                    if (props == null)
                    {
                        props = new ParagraphProperties();
                        paragraph.PrependChild(props);
                    }

                    props.Justification = new Justification() { Val = JustificationValues.Center };
                }
            }
        }

        public static bool IsFileLocked(string path)
        {
            try
            {
                using (FileStream stream = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.None))
                {
                    return false; 
                }
            }
            catch (IOException)
            {
                return true; 
            }
        }

        private bool AreAllFieldsFilled(UserControl userControl)
        {
            
            foreach (Control control in userControl.Controls)
            {
              
                if (control is TextBox textBox)
                {
                    if (textBox.Name == "TypeTextBox") 
                    {
                        continue;
                    }
                    if (string.IsNullOrWhiteSpace(textBox.Text))
                    {
                        return false; 
                    }
                }

                
                if (control is ComboBox comboBox)
                {
                   
                    if (comboBox.SelectedIndex == -1)
                    {
                        return false;
                    }
                }

          
                if (control is DateTimePicker dateTimePicker)
                {
                    if (string.IsNullOrWhiteSpace(dateTimePicker.Text))
                    {
                        return false;
                    }
                }
            }

            return true; 
        }

        private void LoadSubjectIntoComboBox(ComboBox cmb, int specialtyId)
        {
            cmb.DataSource = null;
            cmb.Items.Clear();
            cmb.Enabled = true; 

            using (SqlConnection connection = new SqlConnection(connectionString))
            {
                try
                {
                    connection.Open();
                    string query = "SELECT SubjectID, SubjectName, Semester FROM Subjects WHERE SpecialtyID = @SpecialtyID ORDER BY SubjectName";
                    SqlDataAdapter adapter = new SqlDataAdapter(query, connection);
                    adapter.SelectCommand.Parameters.AddWithValue("@SpecialtyID", specialtyId);
                    DataTable table = new DataTable();
                    adapter.Fill(table);

                    if (table.Rows.Count > 0)
                    {
                        table.Columns.Add("DisplayMemberWithSemester", typeof(string));
                        foreach (DataRow row in table.Rows)
                        {
                            row["DisplayMemberWithSemester"] = $"{row["SubjectName"]} ({row["Semester"]})";
                        }
                        cmb.DisplayMember = "DisplayMemberWithSemester";
                        cmb.ValueMember = "SubjectID";
                        cmb.DataSource = table;
                        cmb.SelectedIndex = 0;
                    }
                    else
                    {
                        cmb.Text = "Нет предметов для спец-ти";
                        cmb.Enabled = false;
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка загрузки предметов: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    cmb.Enabled = false;

        private void LoadSemesterIntoComboBox(ComboBox cmb, int subjectId)
        {
            cmb.DataSource = null;
            cmb.Items.Clear();
            cmb.Enabled = true; 

            using (SqlConnection connection = new SqlConnection(connectionString))
            {
                try
                {
                    connection.Open();
                    string query = "SELECT Semester FROM Subjects WHERE SubjectID = @SubjectID";
                    SqlDataAdapter adapter = new SqlDataAdapter(query, connection);
                    adapter.SelectCommand.Parameters.AddWithValue("@SubjectID", subjectId);
                    DataTable table = new DataTable(); 
                    adapter.Fill(table);

                    if (table.Rows.Count > 0)
                    {
                        DataTable semesterDisplayTable = new DataTable();
                        semesterDisplayTable.Columns.Add("SemesterValue", typeof(int));
                        semesterDisplayTable.Columns.Add("SemesterDisplay", typeof(string));

                        foreach (DataRow row in table.Rows) 
                        {
                            semesterDisplayTable.Rows.Add(Convert.ToInt32(row["Semester"]), row["Semester"].ToString());
                        }

                        cmb.DisplayMember = "SemesterDisplay";
                        cmb.ValueMember = "SemesterValue";
                        cmb.DataSource = semesterDisplayTable;
                        if (semesterDisplayTable.Rows.Count > 0) cmb.SelectedIndex = 0;
                    }
                    else
                    {
                        cmb.Text = "Семестр не найден"; 
                        cmb.Enabled = false;
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка загрузки семестров: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    cmb.Enabled = false;
                }
            }

            }

            checkedListBoxTeachers.Enabled = true;
            string query = "SELECT TeacherID, FullName FROM Teachers WHERE SubjectID = @SubjectID ORDER BY FullName";
            DataTable table = new DataTable();

            try
            {
                using (SqlConnection conn = new SqlConnection(_currentConnectionString))
                using (SqlCommand cmd = new SqlCommand(query, conn))
                {
                    cmd.Parameters.AddWithValue("@SubjectID", subjectId);
                    SqlDataAdapter adapter = new SqlDataAdapter(cmd);
                    adapter.Fill(table); 
                }

                if (table.Rows.Count > 0)
                {
                    Debug.WriteLine($"Найдено преподавателей: {table.Rows.Count} для SubjectID: {subjectId}");
                    checkedListBoxTeachers.DataSource = table;
                    checkedListBoxTeachers.DisplayMember = "FullName";
                    checkedListBoxTeachers.ValueMember = "TeacherID";
                }
                else
                {
                    Debug.WriteLine($"Преподаватели для SubjectID: {subjectId} не найдены.");

                    checkedListBoxTeachers.Items.Add("Для этого предмета нет преподавателей");

                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка загрузки преподавателей для предмета: {ex.Message}", "Ошибка БД", MessageBoxButtons.OK, MessageBoxIcon.Error);
                Debug.WriteLine($"Ошибка SQL при загрузке преподавателей для SubjectID: {subjectId}: {ex.Message}");
                checkedListBoxTeachers.Enabled = false;
            }
            TicketGenerator.UpdateRichTextBox(Properties.Resources.TF5UC, vrd, richTextBox1); 
        }

            {
                string savedTicketFilePath = saveFileDialog.FileName;

                string selectedExtension = Path.GetExtension(saveFileDialog.FileName).ToLower();

            string connectionString = _form2.ConnectionString; // Используем ConnectionString из экземпляра _form2

            int? ticketFileId = DatabaseHelper.GetOrCreateDocumentAndReturnId(

            }

            string specialtyNumberText = Form2.Specialitynumber;

                    return;

        private void LoadData(string query, DataGridView dgv, Dictionary<string, string> columnHeaders, string dateColumnFormat = null, List<string> fileNameColumns = null)
        {
            using (SqlConnection connection = new SqlConnection(connectionString))
            {
                try
                {
                    connection.Open();
                    SqlDataAdapter adapter = new SqlDataAdapter(query, connection);
                    DataTable table = new DataTable();
                    adapter.Fill(table);

                    if (fileNameColumns != null)
                    {
                        foreach (string filePathColumnName in fileNameColumns)
                        {
                            if (table.Columns.Contains(filePathColumnName))
                            {
                                string displayColumnName = filePathColumnName + "Display";
                                if (!table.Columns.Contains(displayColumnName)) 
                                {
                                    table.Columns.Add(displayColumnName, typeof(string));
                                    foreach (DataRow row in table.Rows)
                                    {
                                        if (row[filePathColumnName] != DBNull.Value)
                                        {
                                            row[displayColumnName] = Path.GetFileName(row[filePathColumnName].ToString());
                                        }
                                        else
                                        {
                                            row[displayColumnName] = string.Empty;
                                        }
                                    }
                                }
                            }
                        }
                    }


                    dgv.DataSource = table;
                    dgv.RowHeadersVisible = false;
                    dgv.AllowUserToAddRows = false;
                    dgv.AllowUserToDeleteRows = false;
                    dgv.ReadOnly = true;
                    dgv.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
                    dgv.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill;

                    foreach (DataGridViewColumn column in dgv.Columns)
                    {
                        if (columnHeaders.ContainsKey(column.Name))
                        {
                            if (string.IsNullOrEmpty(columnHeaders[column.Name]))
                            {
                                column.Visible = false;
                            }
                            else
                            {
                                column.HeaderText = columnHeaders[column.Name];
                            }
                        }
                        else if (!column.Name.EndsWith("Display") && fileNameColumns != null && fileNameColumns.Contains(column.Name))
                        {
                            column.Visible = false;
                        }

                    }

                    if (!string.IsNullOrEmpty(dateColumnFormat) && dgv.Columns.Contains(dateColumnFormat.Split(':')[0]))
                    {
                        string colName = dateColumnFormat.Split(':')[0];
                        string format = dateColumnFormat.Split(':')[1];
                        dgv.Columns[colName].DefaultCellStyle.Format = format;
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка загрузки данных для {dgv.Name}: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }

        private void LoadExamEvents()
        {
            string query = @"
                SELECT ev.EventID, s.SubjectName, sp.SpecialtyName, t.FullName AS TeacherFullName,
                       qd.FilePath AS QuestionFilePath, td.FilePath AS TicketFilePath, ev.ExamDate,
                       ev.SubjectID, ev.SpecialtyID, ev.TeacherID, ev.QuestionID, ev.TicketID
                FROM ExamEvents ev
                INNER JOIN Subjects s ON ev.SubjectID = s.SubjectID
                INNER JOIN Specialties sp ON ev.SpecialtyID = sp.SpecialtyID

        private bool CheckUsage(string tableName, string columnName, int id, string messageIfUsed)
        {
            using (SqlConnection connection = new SqlConnection(connectionString))
            {
                try
                {
                    connection.Open();
                    string query = $"SELECT COUNT(*) FROM {tableName} WHERE {columnName} = @id";
                    SqlCommand cmd = new SqlCommand(query, connection);
                    cmd.Parameters.AddWithValue("@id", id);
                    int count = (int)cmd.ExecuteScalar();
                    if (count > 0)
                    {
                        MessageBox.Show(messageIfUsed, "Ошибка удаления", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        return true;
                    }
                    return false;
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка проверки связей: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return true; 
                }
            }

        public static bool SaveDocumentPathToDatabase(string filePath, string tableName, string connectionString)
        {
            if (string.IsNullOrEmpty(filePath))
            {
                MessageBox.Show($"Путь к файлу для таблицы {tableName} не указан.", "Ошибка данных", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }

            string query = $"INSERT INTO {tableName} (FilePath) VALUES (@FilePath)";

            try
            {
                using (SqlConnection conn = new SqlConnection(connectionString))
                using (SqlCommand cmd = new SqlCommand(query, conn))
                {
                    cmd.Parameters.AddWithValue("@FilePath", filePath);
                    conn.Open();
                    cmd.ExecuteNonQuery();
                    return true;
                }
            }
            catch (SqlException ex)
            {
                if (ex.Number == 2627 || ex.Number == 2601) // Коды ошибок для нарушения UNIQUE constraint
                {
                    MessageBox.Show($"Файл '{Path.GetFileName(filePath)}' уже зарегистрирован в базе данных (таблица: {tableName}). Повторное добавление не требуется.", "Информация", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    return true;
                }
                else
                {
                    MessageBox.Show($"Ошибка при сохранении пути в БД (таблица: {tableName}):\n{ex.Message}", "Ошибка базы данных", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return false;
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Общая ошибка при сохранении пути в БД (таблица: {tableName}):\n{ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }
        }
        public static bool SaveExamEvent(int subjectId, int specialtyId, int teacherId, int questionId, int ticketId, DateTime examDate, string connectionString)
        {
            if (subjectId <= 0 || specialtyId <= 0 || teacherId <= 0 || questionId <= 0 || ticketId <= 0)
            {
                MessageBox.Show("Некорректные ID для создания события экзамена.", "Ошибка данных", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }

            string query = @"
            INSERT INTO ExamEvents 
                (SubjectID, SpecialtyID, TeacherID, QuestionID, TicketID, ExamDate)
            VALUES 
                (@SubjectID, @SpecialtyID, @TeacherID, @QuestionID, @TicketID, @ExamDate)";

            try
            {
                using (SqlConnection conn = new SqlConnection(connectionString))
                using (SqlCommand cmd = new SqlCommand(query, conn))
                {
                    cmd.Parameters.AddWithValue("@SubjectID", subjectId);
                    cmd.Parameters.AddWithValue("@SpecialtyID", specialtyId);
                    cmd.Parameters.AddWithValue("@TeacherID", teacherId);
                    cmd.Parameters.AddWithValue("@QuestionID", questionId);
                    cmd.Parameters.AddWithValue("@TicketID", ticketId);
                    cmd.Parameters.AddWithValue("@ExamDate", examDate.Date); // Сохраняем только дату

                    conn.Open();
                    int rowsAffected = cmd.ExecuteNonQuery();
                    return rowsAffected > 0; // Успешно, если хотя бы одна строка была добавлена
                }
            }
            catch (SqlException ex)
            {
                MessageBox.Show($"Ошибка SQL при сохранении события экзамена:\n{ex.Message}", "Ошибка базы данных", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Общая ошибка при сохранении события экзамена:\n{ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }
        }

                }

        private void OpenFile(string filePath)
        {
            if (!string.IsNullOrEmpty(filePath))
            {
                if (File.Exists(filePath))
                {
                    try
                    {
                        Process.Start(new ProcessStartInfo(filePath) { UseShellExecute = true });
                    }
                    catch (Exception ex)
                    {
                        MessageBox.Show($"Не удалось открыть файл: {ex.Message}\nПуть: {filePath}", "Ошибка открытия файла", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
                else
                {
                    MessageBox.Show($"Файл не найден по указанному пути:\n{filePath}", "Файл не найден", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                }
            }
            else
            {
                MessageBox.Show("Путь к файлу не указан для этой записи.", "Информация", MessageBoxButtons.OK, MessageBoxIcon.Information);
            }
        }

        {

            {
                MessageBox.Show($"Путь к файлу для таблицы {tableName} не указан.", "Ошибка данных", MessageBoxButtons.OK, MessageBoxIcon.Warning);

            {
                using (SqlConnection conn = new SqlConnection(connectionString))

                        existingId = cmdSelect.ExecuteScalar();
                    }

                    if (existingId != null && existingId != DBNull.Value)
                    {
                        return Convert.ToInt32(existingId);
                    }
                    else
                    {
                        using (SqlCommand cmdInsert = new SqlCommand(insertQuery, conn))
                        {
                            cmdInsert.Parameters.AddWithValue("@FilePath", filePath);
                            object newId = cmdInsert.ExecuteScalar();
                            if (newId != null && newId != DBNull.Value)
                            {
                                return Convert.ToInt32(newId);
                            }
                            else
                            {
                                MessageBox.Show($"Не удалось получить ID для новой записи в таблице {tableName}.", "Ошибка базы данных", MessageBoxButtons.OK, MessageBoxIcon.Error);

        public static int? GetOrCreateTeacherAndReturnId(string fullName, int subjectId, string connectionString)
        {
            if (string.IsNullOrWhiteSpace(fullName) || subjectId <= 0)
            {
                // Не показываем MessageBox здесь, чтобы не спамить, если вызывается в цикле.
                // Вызывающий код должен обработать null.
                return null;
            }

            fullName = fullName.Trim();
            string selectQuery = "SELECT TeacherID FROM Teachers WHERE FullName = @FullName AND SubjectID = @SubjectID";
            object teacherIdObj;

            try
            {
                using (SqlConnection conn = new SqlConnection(connectionString))
                {
                    conn.Open();
                    // 1. Попытка найти существующего преподавателя для данного предмета
                    using (SqlCommand cmdSelect = new SqlCommand(selectQuery, conn))

                        cmdSelect.Parameters.AddWithValue("@FullName", fullName);
                        cmdSelect.Parameters.AddWithValue("@SubjectID", subjectId);
                        teacherIdObj = cmdSelect.ExecuteScalar();
                    }

                    if (teacherIdObj != null && teacherIdObj != DBNull.Value)
                    {
                        return Convert.ToInt32(teacherIdObj);
                    }
                    else
                    {
                        string insertQuery = "INSERT INTO Teachers (FullName, SubjectID) VALUES (@FullName, @SubjectID); SELECT SCOPE_IDENTITY();";
                        using (SqlCommand cmdInsert = new SqlCommand(insertQuery, conn))
                        {
                            cmdInsert.Parameters.AddWithValue("@FullName", fullName);
                            cmdInsert.Parameters.AddWithValue("@SubjectID", subjectId);
                            object newTeacherIdObj = cmdInsert.ExecuteScalar();
                            if (newTeacherIdObj != null && newTeacherIdObj != DBNull.Value)
                            {
                                Debug.WriteLine($"Добавлен новый преподаватель: {fullName} для предмета ID {subjectId}, TeacherID: {newTeacherIdObj}");
                                return Convert.ToInt32(newTeacherIdObj);
                            }
                            else
                            {

                                Debug.WriteLine($"Не удалось добавить преподавателя '{fullName}' для предмета ID {subjectId} и получить его ID.");
                                return null;
                            }
                        }
                    }
                }
            }
            catch (SqlException ex)
            {
                if (ex.Number == 2627 || ex.Number == 2601)
                {
                    using (SqlConnection connRetry = new SqlConnection(connectionString))
                    {
                        connRetry.Open();
                        using (SqlCommand cmdSelectRetry = new SqlCommand(selectQuery, connRetry))
                        {
                            cmdSelectRetry.Parameters.AddWithValue("@FullName", fullName);
                            cmdSelectRetry.Parameters.AddWithValue("@SubjectID", subjectId);
                            teacherIdObj = cmdSelectRetry.ExecuteScalar();
                            if (teacherIdObj != null && teacherIdObj != DBNull.Value) return Convert.ToInt32(teacherIdObj);
                        }
                    }
                    Debug.WriteLine($"Преподаватель '{fullName}' для предмета ID {subjectId} уже существует (код {ex.Number}). Не удалось получить ID при повторной попытке.");
                    return null;
                }
                MessageBox.Show($"Ошибка SQL при работе с таблицей Teachers для ФИО '{fullName}' и предмета ID {subjectId}:\n{ex.Message}", "Ошибка базы данных", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return null;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Общая ошибка при поиске/добавлении преподавателя '{fullName}' для предмета ID {subjectId}:\n{ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return null;
            }

                    else
                    {
                        if (_initialSubjectIdToSelect != -1)
                        {
                            subjectComboBox.SelectedValue = _initialSubjectIdToSelect;
                            _initialSubjectIdToSelect = -1; 
                        }
                        if (subjectComboBox.SelectedIndex == -1) 
                            subjectComboBox.SelectedIndex = 0;
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка загрузки предметов: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    subjectComboBox.Enabled = false;
                }
            }
        }

        private void LoadFileDocumentComboBox(ComboBox cmb, string query, string valueMemberIdColumn)
        {
             try
            {
                using (SqlConnection conn = new SqlConnection(_connectionString))
                {
                    SqlDataAdapter adapter = new SqlDataAdapter(query, conn);
                    DataTable dt = new DataTable();
                    adapter.Fill(dt);

                    DataTable displayDt = new DataTable();
                    displayDt.Columns.Add(valueMemberIdColumn, typeof(int));
                    displayDt.Columns.Add("DisplayFileName", typeof(string));

                    foreach (DataRow row in dt.Rows)
                    {
                        if (row["FilePath"] != DBNull.Value)
                            displayDt.Rows.Add(row[valueMemberIdColumn], Path.GetFileName(row["FilePath"].ToString()));
                        else
                             displayDt.Rows.Add(row[valueMemberIdColumn], "N/A"); 
                    }

                    cmb.DataSource = displayDt;
                    cmb.DisplayMember = "DisplayFileName";
                    cmb.ValueMember = valueMemberIdColumn;
                    if (displayDt.Rows.Count > 0) cmb.SelectedIndex = 0; else cmb.SelectedIndex = -1;
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка загрузки документов для {cmb.Name}: {ex.Message}", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

            }
        }


       
        
        private void SaveButton_Click(object sender, EventArgs e)
        {
            if (specialtyComboBox.SelectedValue == null ||
                 subjectComboBox.SelectedValue == null || 
                 teacherComboBox.SelectedValue == null ||
                 questionDocumentComboBox.SelectedValue == null ||
                 ticketDocumentComboBox.SelectedValue == null)
            {
                MessageBox.Show("Пожалуйста, выберите все значения для экзамена.", "Ошибка валидации", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            string query;
            if (_eventId.HasValue)
            {
                query = @"UPDATE ExamEvents
                          SET SubjectID=@subId, SpecialtyID=@specId, TeacherID=@teachId,
                              QuestionID=@questId, TicketID=@tickId, ExamDate=@date
                          WHERE EventID=@id";
            }
            else
            {
                query = @"INSERT INTO ExamEvents (SubjectID, SpecialtyID, TeacherID, QuestionID, TicketID, ExamDate)
                          VALUES (@subId, @specId, @teachId, @questId, @tickId, @date)";
            }

            using (SqlConnection conn = new SqlConnection(_connectionString))
            using (SqlCommand cmd = new SqlCommand(query, conn))
            {
                cmd.Parameters.AddWithValue("@specId", (int)specialtyComboBox.SelectedValue);
                cmd.Parameters.AddWithValue("@subId", (int)subjectComboBox.SelectedValue); 
                cmd.Parameters.AddWithValue("@teachId", (int)teacherComboBox.SelectedValue);
                cmd.Parameters.AddWithValue("@questId", (int)questionDocumentComboBox.SelectedValue);
                cmd.Parameters.AddWithValue("@tickId", (int)ticketDocumentComboBox.SelectedValue);
                cmd.Parameters.AddWithValue("@date", examDateTimePicker.Value.Date);
                if (_eventId.HasValue)
                {
                    cmd.Parameters.AddWithValue("@id", _eventId.Value);
                }

                try
                {
                    conn.Open();
                    cmd.ExecuteNonQuery();
                    this.DialogResult = DialogResult.OK;
                    this.Close();
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Ошибка сохранения экзамена: " + ex.Message, "Ошибка базы данных", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }
